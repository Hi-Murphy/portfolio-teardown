def lin(h):
  c=[int(h[i:i+2],16)/255 for i in (0,2,4)]
  return [x/12.92 if x<=0.04045 else ((x+0.055)/1.055)**2.4 for x in c]
def lum(c): return 0.2126*c[0]+0.7152*c[1]+0.0722*c[2]
def cr(a,b):
  la,lb=sorted([lum(a),lum(b)],reverse=True); return (la+0.05)/(lb+0.05)
def lab(c):
  M=[[0.4124,0.3576,0.1805],[0.2126,0.7152,0.0722],[0.0193,0.1192,0.9505]]
  X,Y,Z=[sum(M[i][j]*c[j] for j in range(3)) for i in range(3)]
  n=[0.95047,1,1.08883]; f=lambda t: t**(1/3) if t>0.008856 else 7.787*t+16/116
  fx,fy,fz=f(X/n[0]),f(Y/n[1]),f(Z/n[2]); return (116*fy-16,500*(fx-fy),200*(fy-fz))
def de(a,b): return math.dist(lab(a),lab(b))
def mix(a,b,t): return [a[i]+(b[i]-a[i])*t for i in range(3)]
