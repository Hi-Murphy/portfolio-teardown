# ---- source of the temporary sweep test ----
import { test } from 'vitest';
import { writeFileSync } from 'node:fs';
import { parseProject, parseTechRegistry } from '../src/index';
import { pick, rayBoxDistance, type Ray } from '../src/interaction/pick';
import { fitZoom, layoutProject, VIEW, type Vec3 } from '../src/layout';
import { projectAJson, projectBJson, projectCJson, techRegistryJson } from './fixtures';

const registry = parseTechRegistry(techRegistryJson);
const PROJECTS = [
  ['A', parseProject(projectAJson, { registry })],
  ['B', parseProject(projectBJson, { registry })],
  ['C', parseProject(projectCJson, { registry })],
] as const;
const GAPS = [1.1, 1.4, 1.6, 1.8, 2.0, 2.4];
const STEP = 0.015; // world units in the screen plane
const lines: string[] = [];

function basis(az: number) {
  const se = Math.sin(VIEW.elevation), ce = Math.cos(VIEW.elevation);
  const toCam: Vec3 = [ce * Math.sin(az), se, ce * Math.cos(az)];
  const right: Vec3 = [Math.cos(az), 0, -Math.sin(az)];
  const up: Vec3 = [-se * Math.sin(az), ce, -se * Math.cos(az)];
  return { toCam, right, up };
}

test('S06 RFC occlusion sweep', () => {
  lines.push(`grid step ${STEP} world units; visible = rays whose nearest hit (layer slabs occlude) is the component; full = rays hitting the component box alone; exploded (progress 1)`);
  for (const [name, project] of PROJECTS) {
    for (const gap of GAPS) {
      const layout = layoutProject(project, { explodedGap: gap });
      const progress = new Float32Array(layout.layers.length + layout.components.length).fill(1);
      const b = layout.bounds.exploded;
      const zMobile = fitZoom(b, { width: 360, height: 640 });
      const zDesk = fitZoom(b, { width: 1280, height: 720 });
      const rows: string[] = [];
      let worstAll = Infinity, worstAllId = '';
      for (const azDeg of [45, 135, 225, 315]) {
        const az = (azDeg * Math.PI) / 180;
        const { toCam, right, up } = basis(az);
        const R = Math.hypot(b.max[0] - b.min[0], b.max[1] - b.min[1], b.max[2] - b.min[2]);
        const vis = new Float64Array(layout.components.length);
        const full = new Float64Array(layout.components.length);
        for (let u = -R; u <= R; u += STEP) for (let v = -R; v <= R; v += STEP) {
          const origin: Vec3 = [
            u * right[0] + v * up[0] + 2 * R * toCam[0],
            u * right[1] + v * up[1] + 2 * R * toCam[1],
            u * right[2] + v * up[2] + 2 * R * toCam[2],
          ];
          const ray: Ray = { origin, direction: [-toCam[0], -toCam[1], -toCam[2]] };
          const hit = pick(ray, layout, progress);
          if (hit !== null) vis[hit] += STEP * STEP;
          layout.components.forEach((c, i) => {
            if (rayBoxDistance(ray, c.exploded, c.size) !== null) full[i] += STEP * STEP;
          });
        }
        let worst = Infinity, worstId = '';
        const hidden: string[] = [];
        layout.components.forEach((c, i) => {
          const frac = vis[i] / full[i];
          if (frac < worst) { worst = frac; worstId = c.id; }
          if (frac < 0.5) hidden.push(`${c.id}=${(frac * 100).toFixed(0)}%`);
        });
        if (worst < worstAll) { worstAll = worst; worstAllId = `${worstId}@${azDeg}`; }
        rows.push(`    az ${String(azDeg).padStart(3)}: min visible ${(worst * 100).toFixed(0).padStart(3)}% (${worstId}); <50%: ${hidden.join(', ') || 'none'}`);
        if (azDeg === 45 && (name === 'A' || name === 'C')) {
          const watch = name === 'A' ? ['tide-repository', 'tide-view-model', 'tide-feed', 'live-tide-client', 'tide-cache'] : ['booking-service', 'notifier', 'retry-scheduler', 'slot-planner', 'audit-trail', 'locker-registry', 'dead-letter-store', 'push-gateway', 'locker-hardware'];
          rows.push('      45° named: ' + watch.map((id) => { const i = layout.components.findIndex((c) => c.id === id); return `${id} ${(vis[i] / full[i] * 100).toFixed(0)}%`; }).join(' · '));
        }
      }
      lines.push(`${name} gap ${gap.toFixed(1)}: fitZoom 360x640 ${zMobile.toFixed(1)} px/u, 1280x720 ${zDesk.toFixed(1)} px/u; worst over 4 azimuths ${(worstAll * 100).toFixed(0)}% (${worstAllId})`);
      lines.push(...rows);
    }
  }
  writeFileSync('/tmp/s06/occlusion.txt', lines.join('\n') + '\n');
}, 600000);
